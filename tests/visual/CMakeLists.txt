cmake_minimum_required(VERSION 3.16)
project(EasyRiderVisualCheck LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
if(NOT EASYRIDER_SOURCE_DIR)
    get_filename_component(EASYRIDER_SOURCE_DIR "../.." ABSOLUTE)
endif()
file(TO_CMAKE_PATH "${EASYRIDER_SOURCE_DIR}" EASYRIDER_SOURCE_DIR)
set(CMAKE_AUTOUIC_SEARCH_PATHS "${EASYRIDER_SOURCE_DIR}/ui" "${EASYRIDER_SOURCE_DIR}/ui/widgets")
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
file(GLOB GUI "${EASYRIDER_SOURCE_DIR}/src/gui/*.cpp" "${EASYRIDER_SOURCE_DIR}/src/gui/*.h")
file(GLOB CORE "${EASYRIDER_SOURCE_DIR}/src/core/*.cpp" "${EASYRIDER_SOURCE_DIR}/src/nav/*.cpp" "${EASYRIDER_SOURCE_DIR}/src/nav/*.h")
file(GLOB TEST_UI "${EASYRIDER_SOURCE_DIR}/tests/Test*.cpp" "${EASYRIDER_SOURCE_DIR}/tests/Test*.h" "${EASYRIDER_SOURCE_DIR}/tests/categories/*.cpp")
list(FILTER TEST_UI EXCLUDE REGEX "test_main.cpp$")
add_executable(EasyRiderVisualCheck visual_check.cpp ${GUI} ${CORE} ${TEST_UI})
if(EXISTS "${EASYRIDER_SOURCE_DIR}/assets/controls.qrc")
    target_sources(EasyRiderVisualCheck PRIVATE "${EASYRIDER_SOURCE_DIR}/assets/controls.qrc")
endif()
target_include_directories(EasyRiderVisualCheck PRIVATE "${EASYRIDER_SOURCE_DIR}/src" "${EASYRIDER_SOURCE_DIR}/tests")
target_link_libraries(EasyRiderVisualCheck PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)

# Same executable can be built against the saved pre-recovery sources.
add_executable(EasyRiderViewProbe view_probe.cpp ${GUI} ${CORE})
target_include_directories(EasyRiderViewProbe PRIVATE "${EASYRIDER_SOURCE_DIR}/src")
target_link_libraries(EasyRiderViewProbe PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
